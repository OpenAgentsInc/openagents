// Public RSA verification arithmetic. This does not admit an FI assertion.
#load "sha256.coax";
#authority;
RsaWords :: struct { words:[128]u32; }
RSA_BASE :: cast(u64)4294967296;
RSA_MASK :: cast(u64)4294967295;
#test "public modular add and multiply reduce exact boundaries" {
 n:RsaWords;n.words[0]=17;a:RsaWords;b:RsaWords;
 a.words[0]=16;b.words[0]=16;
 #assert rsa_add_mod(a,b,n,1).words[0]==15;
 #assert rsa_mul_mod(a,b,n,1).words[0]==1;
 a.words[0]=0;#assert rsa_mul_mod(a,b,n,1).words[0]==0;
 a.words[0]=12;b.words[0]=13;#assert rsa_mul_mod(a,b,n,1).words[0]==3;
 n.words[0]=4294967291;a.words[0]=4294967290;b.words[0]=4294967290;
 #assert rsa_add_mod(a,b,n,1).words[0]==4294967289;
 #assert rsa_mul_mod(a,b,n,1).words[0]==1;
}

// Inputs to the internal operations are reduced and count is 1..128.
// One u32 limb plus one u32 limb plus a carry fits in u64.
rsa_add_mod :: (a:RsaWords,b:RsaWords,n:RsaWords,count:int)->RsaWords {
 sum:RsaWords;difference:RsaWords;carry:=cast(u64)0;
 for i:0..<count {
  wide:=cast(u64)a.words[i]+cast(u64)b.words[i]+carry;
  sum.words[i]=cast(u32)(wide&RSA_MASK);carry=wide>>32;
 }
 borrow:=cast(u64)0;
 for i:0..<count {
  wide:=RSA_BASE+cast(u64)sum.words[i]-cast(u64)n.words[i]-borrow;
  difference.words[i]=cast(u32)(wide&RSA_MASK);borrow=1-(wide>>32);
 }
 if carry!=0 || borrow==0 {return difference;}
 return sum;
}
rsa_mul_mod :: (a:RsaWords,b:RsaWords,n:RsaWords,count:int)->RsaWords {
 result:RsaWords;power:=a;
 // All operands, branches and exponents here are public.
 for i:0..<count {
  for bit:0..<32 {
   if ((b.words[i]>>bit)&1)!=0 {result=rsa_add_mod(result,power,n,count);}
   power=rsa_add_mod(power,power,n,count);
  }
 }
 return result;
}
rsa_load :: (bytes:[]u8)->RsaWords {
 result:RsaWords;
 for i:0..<bytes.count {
  index:=bytes.count-1-i;
  result.words[i/4]=result.words[i/4]|(cast(u32)bytes[index]<<((i%4)*8));
 }
 return result;
}
rsa_less :: (a:RsaWords,b:RsaWords,count:int)->bool {
 for offset:0..<count {
  i:=count-1-offset;
  if a.words[i]<b.words[i] {return true;}
  if a.words[i]>b.words[i] {return false;}
 }
 return false;
}
rsa_pkcs1_sha256 :: (em:[]u8,digest:[]u8)->bool {
 if em.count<62 || digest.count!=32 {return false;}
 if em[0]!=0 || em[1]!=1 {return false;}
 separator:=em.count-52;
 if separator<10 {return false;}
 for i:2..<separator {if em[i]!=255 {return false;}}
 if em[separator]!=0 {return false;}
 prefix:=[19]u8.[48,49,48,13,6,9,96,134,72,1,101,3,4,2,1,5,0,4,32];
 for i:0..<19 {if em[separator+1+i]!=prefix[i] {return false;}}
 for i:0..<32 {if em[separator+20+i]!=digest[i] {return false;}}
 return true;
}
// The selected public exponent is fixed to 65537; callers cannot choose an algorithm.
rsa_verify_sha256 :: (modulus:[]u8,signature:[]u8,message:[]u8)->bool #export_c {
 if modulus.count<256 || modulus.count>512 || signature.count!=modulus.count {return false;}
 if modulus[0]==0 || (modulus.count==256 && modulus[0]<128) || (modulus[modulus.count-1]&1)==0 {return false;}
 n:=rsa_load(modulus);s:=rsa_load(signature);count:=(modulus.count+3)/4;
 if !rsa_less(s,n,count) {return false;}
 digest:[32]u8;if sha256_hash_checked(message,digest[..],false)!=SHA_OK {return false;}
 // 65537 = 2^16 + 1; 16 squarings and one multiply.
 value:=s;for i:0..<16 {value=rsa_mul_mod(value,value,n,count);}
 value=rsa_mul_mod(value,s,n,count);
 encoded:[512]u8;
 for i:0..<modulus.count {encoded[modulus.count-1-i]=cast(u8)((value.words[i/4]>>((i%4)*8))&255);}
 return rsa_pkcs1_sha256(encoded[0..<modulus.count],digest[..]);
}
