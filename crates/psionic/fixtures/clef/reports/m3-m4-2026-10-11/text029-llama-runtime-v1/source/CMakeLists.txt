cmake_minimum_required(VERSION 3.20)
project(clef_llama_diagnostic LANGUAGES C CXX)
if(NOT DEFINED CLEF_UPSTREAM_SOURCE)
    message(FATAL_ERROR "Explicit pinned upstream source is required")
endif()
add_subdirectory("${CLEF_UPSTREAM_SOURCE}" upstream)
add_executable(clef-llama-diagnostic driver.cpp)
target_include_directories(clef-llama-diagnostic PRIVATE "${CLEF_UPSTREAM_SOURCE}/src")
target_compile_features(clef-llama-diagnostic PRIVATE cxx_std_17)
target_link_libraries(clef-llama-diagnostic PRIVATE llama ggml-cpu)
