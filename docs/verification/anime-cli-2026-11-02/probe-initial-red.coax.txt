acp_catalogue :: (s: string) -> bool {
  nodes: [512] JvNode; d: [1] Jv;
  if !jv_parse(d[..],s,nodes[..]) || jv_int(d[0],jv_get(d[0],0,"v"),0) != 1 || jv_raw(d[0],jv_get(d[0],0,"schema")) != "anime.catalogue.v1" || jv_raw(d[0],jv_get(d[0],0,"stage")) != "courtyard" { return false; }
  cast_list := jv_get(d[0],0,"cast"); kinds := jv_path(d[0],0,"questions.kind");
  if jv_kind(d[0],cast_list) != JV_ARRAY || jv_len(d[0],cast_list) != 7 || jv_kind(d[0],kinds) != JV_ARRAY || jv_len(d[0],kinds) != 10 { return false; }
  expected := [10] string.["enter","exit","act","emote","cast","fx","shot","cut","beat","make"];
  for i: 0..<10 { if jv_raw(d[0],jv_at(d[0],kinds,i)) != expected[i] { return false; } }
  for i: 0..<7 { row := jv_at(d[0],cast_list,i); if jv_int(d[0],jv_get(d[0],row,"slot"),-1) != i || jv_kind(d[0],jv_get(d[0],row,"body_drawn")) != (ifx i == 6 then JV_FALSE else JV_TRUE) { return false; } }
  return true;
}
acp_take :: (s: string, lines: int, answers: int, makeup: bool) -> bool {
  nodes: [4096] JvNode; d: [1] Jv;
  if !jv_parse(d[..],s,nodes[..]) || jv_raw(d[0],jv_get(d[0],0,"schema")) != "anime.v1" { return false; }
  list := jv_get(d[0],0,"lines"); if jv_kind(d[0],list) != JV_ARRAY || jv_len(d[0],list) != lines { return false; }
  total := 0; made := false; previous := -1.0;
  for i: 0..<lines {
    line := jv_at(d[0],list,i); said := args_number(jv_raw(d[0],jv_get(d[0],line,"said"))); done := args_number(jv_raw(d[0],jv_get(d[0],line,"done")));
    if !said.ok || !done.ok || said.value != done.value || said.value < 0.0 || said.value < previous || (i == 0 && said.value != 0.0) { return false; }
    previous = done.value; a := jv_get(d[0],line,"answers"); if jv_kind(d[0],a) != JV_ARRAY { return false; }
    total += jv_len(d[0],a);
    for k: 0..<jv_len(d[0],a) {
      at := jv_at(d[0],a,k); p := args_number(jv_raw(d[0],jv_get(d[0],at,"p")));
      if !p.ok || p.value < 0.0 || p.value > 1.0 || jv_get(d[0],at,"made") >= 0 { return false; }
      if jv_raw(d[0],jv_get(d[0],at,"kind")) == "make" {
        if jv_kind(d[0],jv_get(d[0],at,"stock")) != JV_FALSE || jv_kind(d[0],jv_get(d[0],at,"render_supported")) != JV_FALSE || jv_raw(d[0],jv_get(d[0],at,"decision")) != "heard" { return false; }
        made = true;
      }
    }
  }
  return total == answers && made == makeup;
}
acp_context :: (s: string, words: string, last: int) -> bool {
  nodes: [2048] JvNode; d: [1] Jv; decoded: [601] u8;
  if !jv_parse(d[..],s,nodes[..]) || jv_int(d[0],jv_get(d[0],0,"v"),0) != 1 { return false; }
  n := jv_str(d[0],jv_get(d[0],0,"words"),decoded[..]); if n != words.count || cast(string)decoded[0..<n] != words { return false; }
  cast_list := jv_path(d[0],0,"state.cast"); shot := jv_path(d[0],0,"state.shot"); history := jv_path(d[0],0,"state.last");
  if jv_kind(d[0],cast_list) != JV_ARRAY || jv_len(d[0],cast_list) < 1 || jv_len(d[0],cast_list) > 7 || jv_kind(d[0],shot) != JV_OBJECT || jv_kind(d[0],history) != JV_ARRAY || jv_len(d[0],history) != last { return false; }
  for i: 0..<last { item := jv_at(d[0],history,i); if jv_raw(d[0],jv_get(d[0],item,"kind")) == "make" || jv_get(d[0],item,"confidence") >= 0 || jv_get(d[0],item,"decision") >= 0 { return false; } }
  return true;
}

tool_main :: () -> int #export_c {
  cli: [4096] u8; dir: [4096] u8;
  binary := proc_getenv("ANIME_CLI_BIN",cli[..]); record := proc_getenv("ANIME_CLI_RECORD",dir[..]);
  if binary == "" || record == "" || !fs_mkdir(record) { return 2; }
  h := proc_new(); args := [10] string.[binary,"--json","--profile","anime-probe-red","--origin","http://127.0.0.1:57435","labs","anime","catalogue","--help"];
  ok := h >= 0;
  for arg: args[0..<9] { ok = proc_arg(h,arg) && ok; }
  ok = proc_env(h,"OPENAGENTS_KEYCHAIN=0") && proc_env(h,"OPENAGENTS_NSEC=invalid-test-key") && ok;
  if !ok || proc_start(h) != 0 { return 2; }
  out: [65536] u8; n := proc_line(h,30000,out[..]); status := proc_wait(h,2000); done := proc_release(h);
  path: [4096] u8; wrote := fs_write(fs_join(record,"catalogue-red.cli.json",path[..]),ifx n > 0 then cast(string)out[0..<n] else "{}");
  if status != 0 || n < 1 { print("FAIL built CLI exposes labs anime catalogue (exit %)\n",status); return 1; }
  nodes: [256] JvNode; d: [1] Jv; value: [8192] u8; o: [1] Jo;
  if !jv_parse(d[..],cast(string)out[0..<n],nodes[..]) { return 1; }
  jo_start(o[..],value[..]); jv_write(d[0],jv_get(d[0],0,"catalogue"),o[..]);
  return ifx wrote && acp_catalogue(jo_text(o[..])) then 0 else 1;
}
