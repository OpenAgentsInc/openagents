acp_catalogue :: (s: string) -> bool {
  nodes: [512] JvNode; d: [1] Jv;
  if !jv_parse(d[..],s,nodes[..]) || jv_int(d[0],jv_get(d[0],0,"v"),0) != 1 || jv_raw(d[0],jv_get(d[0],0,"schema")) != "anime.catalogue.v1" || jv_raw(d[0],jv_get(d[0],0,"stage")) != "courtyard" { return false; }
  cast_list := jv_get(d[0],0,"cast"); kinds := jv_path(d[0],0,"questions.kind");
  if jv_kind(d[0],cast_list) != JV_ARRAY || jv_len(d[0],cast_list) != 7 || jv_kind(d[0],kinds) != JV_ARRAY || jv_len(d[0],kinds) != 10 { return false; }
  expected := [10] string.["enter","exit","act","emote","cast","fx","shot","cut","beat","make"];
  for i: 0..<10 { if jv_raw(d[0],jv_at(d[0],kinds,i)) != expected[i] { return false; } }
  for i: 0..<7 { row := jv_at(d[0],cast_list,i); if jv_int(d[0],jv_get(d[0],row,"slot"),-1) != i || jv_kind(d[0],jv_get(d[0],row,"body_drawn")) != (ifx i == 6 then JV_FALSE else JV_TRUE) { return false; } }
  return true;
}
acp_take :: (s: string, lines: int, answers: int, makeup: bool) -> bool {
  nodes: [4096] JvNode; d: [1] Jv;
  if !jv_parse(d[..],s,nodes[..]) || jv_raw(d[0],jv_get(d[0],0,"schema")) != "anime.v1" { return false; }
  list := jv_get(d[0],0,"lines"); if jv_kind(d[0],list) != JV_ARRAY || jv_len(d[0],list) != lines { return false; }
  total := 0; made := false; previous := -1.0;
  for i: 0..<lines {
    line := jv_at(d[0],list,i); said := args_number(jv_raw(d[0],jv_get(d[0],line,"said"))); done := args_number(jv_raw(d[0],jv_get(d[0],line,"done")));
    if !said.ok || !done.ok || said.value != done.value || said.value < 0.0 || said.value < previous || (i == 0 && said.value != 0.0) { return false; }
    previous = done.value; a := jv_get(d[0],line,"answers"); if jv_kind(d[0],a) != JV_ARRAY { return false; }
    total += jv_len(d[0],a);
    for k: 0..<jv_len(d[0],a) {
      at := jv_at(d[0],a,k); p := args_number(jv_raw(d[0],jv_get(d[0],at,"p")));
      if !p.ok || p.value < 0.0 || p.value > 1.0 || jv_get(d[0],at,"made") >= 0 { return false; }
      if jv_raw(d[0],jv_get(d[0],at,"kind")) == "make" {
        if jv_kind(d[0],jv_get(d[0],at,"stock")) != JV_FALSE || jv_kind(d[0],jv_get(d[0],at,"render_supported")) != JV_FALSE || jv_raw(d[0],jv_get(d[0],at,"decision")) != "heard" { return false; }
        made = true;
      }
    }
  }
  return total == answers && made == makeup;
}
acp_context :: (s: string, words: string, last: int) -> bool {
  nodes: [2048] JvNode; d: [1] Jv; decoded: [601] u8;
  if !jv_parse(d[..],s,nodes[..]) || jv_int(d[0],jv_get(d[0],0,"v"),0) != 1 { return false; }
  n := jv_str(d[0],jv_get(d[0],0,"words"),decoded[..]); if n != words.count || cast(string)decoded[0..<n] != words { return false; }
  cast_list := jv_path(d[0],0,"state.cast"); shot := jv_path(d[0],0,"state.shot"); history := jv_path(d[0],0,"state.last");
  if jv_kind(d[0],cast_list) != JV_ARRAY || jv_len(d[0],cast_list) < 1 || jv_len(d[0],cast_list) > 7 || jv_kind(d[0],shot) != JV_OBJECT || jv_kind(d[0],history) != JV_ARRAY || jv_len(d[0],history) != last { return false; }
  for i: 0..<last { item := jv_at(d[0],history,i); if jv_raw(d[0],jv_get(d[0],item,"kind")) == "make" || jv_get(d[0],item,"confidence") >= 0 || jv_get(d[0],item,"decision") >= 0 { return false; } }
  return true;
}
acp_report :: (reply: string, take: string, words: string, lines: int, total: int, count: int) -> bool {
  rn: [4096] JvNode; tn: [4096] JvNode; r: [1] Jv; d: [1] Jv; heard: [601] u8;
  if !jv_parse(r[..],reply,rn[..]) || !jv_parse(d[..],take,tn[..]) { return false; }
  n := jv_str(r[0],jv_get(r[0],0,"heard"),heard[..]);
  if n != words.count || n < 0 || cast(string)heard[0..<n] != words || jv_int(r[0],jv_get(r[0],0,"lines"),-1) != lines || jv_int(r[0],jv_get(r[0],0,"answer_count"),-1) != total { return false; }
  reports := jv_get(r[0],0,"answers"); last := jv_at(d[0],jv_get(d[0],0,"lines"),lines-1); saved := jv_get(d[0],last,"answers");
  if jv_kind(r[0],reports) != JV_ARRAY || jv_len(r[0],reports) != count || jv_kind(d[0],saved) != JV_ARRAY || jv_len(d[0],saved) != count { return false; }
  return jv_raw(r[0],reports) == jv_raw(d[0],saved);
}
acp_preserved :: (before: string, after: string) -> bool {
  bn: [4096] JvNode; an: [4096] JvNode; b: [1] Jv; a: [1] Jv;
  if !jv_parse(b[..],before,bn[..]) || !jv_parse(a[..],after,an[..]) { return false; }
  old := jv_get(b[0],0,"lines"); next := jv_get(a[0],0,"lines");
  if jv_kind(b[0],old) != JV_ARRAY || jv_kind(a[0],next) != JV_ARRAY || jv_len(a[0],next) < jv_len(b[0],old) { return false; }
  for i: 0..<jv_len(b[0],old) { if jv_raw(b[0],jv_at(b[0],old,i)) != jv_raw(a[0],jv_at(a[0],next,i)) { return false; } }
  return true;
}

acp_put :: (out: [] u8, n: int, s: string) -> int {
  if n < 0 || n+s.count > out.count { return -1; } for i: 0..<s.count { out[n+i] = s[i]; } return n+s.count;
}
acp_path :: (t: [] Acp, name: string, out: [] u8) -> string { return fs_join(t[0].dir,name,out); }
acp_body :: (t: [] Acp) -> string { return cast(string)t[0].out[0..<t[0].n]; }
acp_chrome_sentinel :: (t: [] Acp) -> string { return cast(string)t[0].chrome_sentinel[0..<t[0].chrome_n]; }
acp_has :: (s: string, needle: string) -> bool { return cjf_find(s,needle) >= 0; }
acp_check :: (t: [] Acp, name: string, ok: bool) -> bool {
  if ok { t[0].pass += 1; print("ok %\n",name); } else { t[0].fail += 1; print("FAIL % (CLI exit %)\n",name,t[0].status); } return ok;
}
acp_note :: (t: [] Acp, name: string, ok: bool) { checked := acp_check(t,name,ok); }
acp_record :: (t: [] Acp, name: string, data: string) -> bool { p: [4096] u8; return fs_write(acp_path(t,name,p[..]),data); }
acp_control :: (t: [] Acp, name: string, data: string) -> bool { p: [4096] u8; return fs_write(fs_join(t[0].control,name,p[..]),data); }
acp_control_read :: (t: [] Acp, name: string, out: [] u8) -> int { p: [4096] u8; return fs_read(fs_join(t[0].control,name,p[..]),out); }
acp_mode :: (t: [] Acp, mode: string) -> bool {
  p: [4096] u8; removed := fs_remove(fs_join(t[0].control,"entered",p[..])); removed = fs_remove(fs_join(t[0].control,"release",p[..]));
  return acp_control(t,"mode",mode);
}
acp_calls :: (t: [] Acp) -> int { b: [32] u8; n := acp_control_read(t,"calls",b[..]); return ifx n > 0 then args_whole(cast(string)b[0..<n]) else -1; }
acp_same :: (path: string, expected: string) -> bool {
  b: [32769] u8; n := fs_read(path,b[..]); return n == expected.count && n >= 0 && cast(string)b[0..<n] == expected;

acp_http_reply :: (h: int, code: int, content: string, body: string) -> bool {
  bytes: [256] u8; o: [1] Jo; number: [32] u8; no: [1] Jo; jo_start(o[..],bytes[..]);
  jo_start(no[..],number[..]); jo_int(no[..],code); jo_put(o[..],"HTTP/1.1 "); jo_put(o[..],jo_text(no[..])); jo_put(o[..]," Fixture\r\nContent-Type: "); jo_put(o[..],content);
  jo_put(o[..],"\r\nCache-Control: no-store\r\nConnection: close\r\nContent-Length: "); jo_start(no[..],number[..]); jo_int(no[..],body.count); jo_put(o[..],jo_text(no[..])); jo_put(o[..],"\r\n\r\n");
  for c: jo_text(o[..]) { if cxsock_write(h,cast(int)c) != 0 { return false; } }
  for c: body { if cxsock_write(h,cast(int)c) != 0 { return false; } }
  return cxsock_flush(h,2000) == 0;
}
// Browser-only glue representing an old page: readiness and a fixed-step hook
// exist, while the take capability is absent. The Coax caller decides refusal.
ACP_OLD_PAGE :: "<!doctype html><canvas width='640' height='360'></canvas><script>globalThis.coaxial={ready:true,sceneReady:true,send:()=>{},probe:()=>0,guardProbe:()=>0};globalThis.coaxialClipStep=async()=>1;</script>";
acp_serve :: (t: [] Acp) {
  if t[0].listener < 0 { return; } h := cxsock_accept(t[0].listener,1); if h < 0 { return; }
  raw: [65536] u8; used := 0; until := proc_now_ms()+2000; complete := false;
  while used < raw.count && proc_now_ms() < until {
    got := cxsock_read(h,50); if got < 0 { break; }
    for i: 0..<got { if used >= raw.count { break; } raw[used] = cast(u8)cxsock_byte(h,i); used += 1; }
    head := cjf_find(cast(string)raw[0..<used],"\r\n\r\n");
    if head >= 0 { length := cjf_header(cast(string)raw[0..<used],"content-length"); want := ifx length == "" then 0 else args_whole(length); if want >= 0 && used == head+4+want { complete = true; break; } }
  }
  request := cjf_request(cast(string)raw[0..<used]);
  if complete && request.ok && request.method == "POST" && request.path == "/api/playtest/labs/anime/direct" {
    t[0].fixture_calls += 1; t[0].fixture_auth = request.has_auth;
    written := acp_record(t,"fixture-request.json",request.body);
    code := ifx t[0].fixture_mode == 0 then 200 else t[0].fixture_mode;
    sent := acp_http_reply(h,code,"application/json",ifx code == 200 then cast(string)t[0].fixture_reply[0..<t[0].fixture_n] else "{\"v\":1,\"error\":\"fixture refusal\"}");
  } else if complete && request.ok && request.method == "GET" { sent := acp_http_reply(h,200,"text/html",ACP_OLD_PAGE); }
  else { sent := acp_http_reply(h,400,"application/json","{}"); }
  closed := cxsock_close(h);
}
acp_cli :: (t: [] Acp, words: [] string, expected: int, text: bool, name: string) -> bool {
  t[0].n = 0; t[0].status = -1; t[0].race_done = false;
  if proc_now_ms() >= t[0].end_ms { return false; }
  h := proc_new(); if h < 0 { return false; }
  ok := proc_arg(h,t[0].cli) && proc_arg(h,"--profile") && proc_arg(h,t[0].profile) && proc_arg(h,"--origin") && proc_arg(h,t[0].origin);
  if !text { ok = proc_arg(h,"--json") && ok; }
  for word: words { ok = proc_arg(h,word) && ok; }
  direct := words.count >= 3 && words[0] == "labs" && words[1] == "anime" && words[2] == "direct";
  ok = proc_env(h,"OPENAGENTS_KEYCHAIN=0") && proc_env(h,ifx t[0].bad_key || !direct then "OPENAGENTS_NSEC=invalid-test-key" else cast(string)t[0].key[0..<t[0].key_n]) && ok;
  if t[0].chrome_n > 0 {
    self: [4096] u8; sn := cxproc_self_path(self[..]); env: [8192] u8;
    en := acp_put(env[..],0,"CHROME="); en = acp_put(env[..],en,cast(string)self[0..<(ifx sn > 0 then sn else 0)]);
    ok = sn > 0 && proc_env(h,cast(string)env[0..<en]) && ok;
    en = acp_put(env[..],0,"ANIME_CLI_CHROME_SENTINEL="); en = acp_put(env[..],en,acp_chrome_sentinel(t)); ok = proc_env(h,cast(string)env[0..<en]) && ok;
  }
  if !ok || proc_start(h) != 0 { done := proc_release(h); return false; }
  until := proc_now_ms()+120000; if t[0].end_ms < until { until = t[0].end_ms; }
  line: [65536] u8; ended := false; cut := false;
  while proc_now_ms() < until {
    acp_serve(t);
    if t[0].race_path != "" && !t[0].race_done {
      entered: [32] u8;
      if acp_control_read(t,"entered",entered[..]) > 0 { t[0].race_done = fs_write(t[0].race_path,cast(string)t[0].race_body[0..<t[0].race_n]) && acp_control(t,"release","yes"); }
    }
    n := proc_line(h,10,line[..]); if n == PROC_EOF { ended = true; break; } if n == PROC_TIMEOUT { continue; } if n < 0 { cut = true; break; }
    if t[0].n > 0 { if t[0].n >= t[0].out.count { cut = true; break; } t[0].out[t[0].n] = '\n'; t[0].n += 1; }
    if t[0].n+n > t[0].out.count { cut = true; break; }
    for i: 0..<n { t[0].out[t[0].n+i] = line[i]; } t[0].n += n;
  }
  if !ended { killed := proc_kill(h,PROC_SIGKILL); }
  t[0].status = proc_wait(h,2000); done := proc_release(h);
  file: [128] u8; fn := acp_put(file[..],0,name); fn = acp_put(file[..],fn,ifx text then ".txt" else ".json");
  written := fn > 0 && acp_record(t,cast(string)file[0..<fn],acp_body(t));
  if !written || !ended || cut || t[0].status != expected { return false; }
  if text { return true; }
  nodes: [2048] JvNode; d: [1] Jv;
  return jv_parse(d[..],acp_body(t),nodes[..]) && jv_kind(d[0],jv_get(d[0],0,"ok")) == (ifx expected == 0 then JV_TRUE else JV_FALSE);
}
