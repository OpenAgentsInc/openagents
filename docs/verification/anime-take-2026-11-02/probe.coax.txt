atp_page_context :: (s: string, lines: int, answers: int, events: int, heard: int, source: int, time: int) -> bool {
  nodes: [64] JvNode; d: [1] Jv;
  if !jv_parse(d[..],s,nodes[..]) || jv_kind(d[0],jv_get(d[0],0,"ready")) != JV_TRUE || jv_int(d[0],jv_get(d[0],0,"width"),0) != 640 || jv_int(d[0],jv_get(d[0],0,"height"),0) != 360 { return false; }
  f := jv_get(d[0],0,"fields"); expected := [8] int.[1,lines,answers,events,8000,heard,source,time];
  if jv_kind(d[0],f) != JV_ARRAY || jv_len(d[0],f) != 8 { return false; }
  for i: 0..<8 { if jv_int(d[0],jv_at(d[0],f,i),-999) != expected[i] { return false; } } return true;
}

atp_b64 :: (s: string, out: [] u8) -> string {
  n := b64_encode(s,out); if n < 0 { return ""; }
  for i: 0..<n { if out[i] == '+' { out[i] = '-'; } else if out[i] == '/' { out[i] = '_'; } }
  while n > 0 && out[n-1] == '=' { n -= 1; } return cast(string)out[0..<n];
}
atp_put :: (out: [] u8, n: int, s: string) -> int {
  if n < 0 || n+s.count > out.count { return -1; }
  for i: 0..<s.count { out[n+i] = s[i]; } return n+s.count;
}
atp_url :: (base: string, take: string, suffix: string, out: [] u8) -> string {
  b: [43696] u8; code := atp_b64(take,b[..]);
  if code == "" { return ""; }
  n := atp_put(out,0,base); n = atp_put(out,n,"/realms/anime-lab?take64="); n = atp_put(out,n,code); n = atp_put(out,n,suffix);
  return ifx n >= 0 then cast(string)out[0..<n] else "";
}
atp_video :: (s: string, w: int, h: int, count: int) -> bool {
  nodes: [256] JvNode; d: [1] Jv;
  if !jv_parse(d[..],s,nodes[..]) { return false; }
  streams := jv_get(d[0],0,"streams");
  if jv_kind(d[0],streams) != JV_ARRAY || jv_len(d[0],streams) != 1 { return false; }
  v := jv_at(d[0],streams,0); dur := args_number(jv_raw(d[0],jv_path(d[0],0,"format.duration")));
  return jv_raw(d[0],jv_get(d[0],v,"codec_type")) == "video" && jv_raw(d[0],jv_get(d[0],v,"codec_name")) == "h264" && jv_raw(d[0],jv_get(d[0],v,"pix_fmt")) == "yuv420p" && jv_int(d[0],jv_get(d[0],v,"width"),0) == w && jv_int(d[0],jv_get(d[0],v,"height"),0) == h && jv_raw(d[0],jv_get(d[0],v,"r_frame_rate")) == "12/1" && args_whole(jv_raw(d[0],jv_get(d[0],v,"nb_read_frames"))) == count && dur.ok && dur.value > cast(f64)count/12.0-0.001 && dur.value < cast(f64)count/12.0+0.001;
}
atp_changed :: (a: [] u8, b: [] u8) -> int {
  if a.count != b.count { return -1; }
  n := 0; for i: 0..<a.count { if a[i] != b[i] { n += 1; } } return n;
}
atp_rgb_error :: (a: [] u8, b: [] u8) -> f64 {
  if a.count != b.count || a.count < 4 || a.count%4 != 0 { return -1.0; }
  sum := 0;
  for i: 0..<a.count { if i%4 != 3 { d := cast(int)a[i]-cast(int)b[i]; sum += ifx d < 0 then -d else d; } }
  return cast(f64)sum/cast(f64)(a.count/4*3);
}

atp_write :: (t: [] Atp, name: string, body: string) -> bool { p: [4096] u8; return fs_write(atp_path(t,name,p[..]),body); }
atp_run :: (t: [] Atp, a: [] Args, name: string, cached: bool) -> bool {
  so: [131072] u8; se: [65536] u8; r := proc_run(a[0].v[0..<a[0].n],atp_timeout(t,120000),so[..],se[..]);
  path: [128] u8; sn := atp_put(path[..],0,name); sn = atp_put(path[..],sn,".cli.json");
  written := atp_write(t,cast(string)path[0..<sn],cast(string)so[0..<r.out_n]);
  sn = atp_put(path[..],0,name); sn = atp_put(path[..],sn,".stderr.log"); written = atp_write(t,cast(string)path[0..<sn],cast(string)se[0..<r.err_n]) && written;
  nodes: [256] JvNode; d: [1] Jv;
  valid := written && r.started == 0 && r.status == 0 && !r.timed_out && jv_parse(d[..],cast(string)so[0..<r.out_n],nodes[..]) && jv_kind(d[0],jv_get(d[0],0,"ok")) == JV_TRUE && jv_kind(d[0],jv_get(d[0],0,"cached")) == (ifx cached then JV_TRUE else JV_FALSE);
  label: [4200] u8; label_n := atp_put(label[..],0,args_at(a,5)); label_n = atp_put(label[..],label_n,".json");
  label_bytes: [65536] u8; ln := ifx label_n > 0 then fs_read(cast(string)label[0..<label_n],label_bytes[..]) else -1;
  label_nodes: [128] JvNode; ld: [1] Jv;
  valid = valid && ln > 0 && jv_parse(ld[..],cast(string)label_bytes[0..<ln],label_nodes[..]);
  if valid {
    page := jv_raw(ld[0],jv_get(ld[0],0,"page"));
    valid = page.count == 71 && page[0..<7] == "sha256:";
    if valid { for c: page[7..<page.count] { valid = ((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f')) && valid; } }
  }
  if !valid || args_at(a,3) != "shot" || name == "shot-default" { return valid; }
  value: [4096] u8; vn := jv_str(d[0],jv_get(d[0],0,"eval"),value[..]); if vn < 0 { return false; }
  without := name == "shot-without-make" || name == "shot-caption-control";
  count := ifx without then 2 else 3; heard := ifx without then 0 else 1;
  bytes := ifx name == "shot-max-a" || name == "shot-max-b" then 32768 else ifx without then ATP_WITHOUT_MAKE.count else ATP_TAKE.count;

  return atp_page_context(cast(string)value[0..<vn],count,count,2,heard,bytes,time);
}

atp_sha :: (data: [] u8, out: [] u8) -> string {
  h: [32] u8; work: [1] ShaOwned; digits := "0123456789abcdef";
  if out.count < 64 || !so_init(work[..]) || !so_update(work[..],data,false) || !so_finish(work[..],h[..],false) { return ""; }
  for i: 0..<32 { out[2*i] = digits[cast(int)(h[i]>>4)]; out[2*i+1] = digits[cast(int)(h[i]&15)]; } return cast(string)out[0..<64];
}
atp_pixels :: (path: string, out: [] u8) -> bool {

  return info.status == PNG_OK && info.w == 640 && info.h == 360 && info.work <= work.count && png_decode(file[0..<n],work[..],out) == PNG_OK;
}

  report: [1024] u8; o: [1] Jo; ah: [64] u8; bh: [64] u8;

}

// The probe itself is a Chrome sentinel for this one negative-control child.
// If the CLI starts Chrome before refusing the oversized canonical cache input,
// the sentinel records that host effect and returns immediately; no browser runs.
atp_input_refusal :: (t: [] Atp, url: string) {
  script: [25000] u8; for i: 0..<script.count { script[i] = 'x'; } script[0] = '"'; script[24999] = '"';
  path: [4096] u8; marker: [4096] u8; self: [4096] u8; sn := cxproc_self_path(self[..]);
  a: [1] Args; ok := args_add(a[..],t[0].cli) && args_add(a[..],"--json") && args_add(a[..],"labs") && args_add(a[..],"shot") && args_add(a[..],url) && args_add(a[..],atp_path(t,"refused.png",path[..]));
  for arg: [4] string.["--eval",cast(string)script[..],"--fresh","--phone"] { ok = args_add(a[..],arg) && ok; }
  h := proc_new(); ok = h >= 0 && sn > 0 && ok;
  for arg: a[0].v[0..<a[0].n] { ok = proc_arg(h,arg) && ok; }
  env: [8192] u8; n := atp_put(env[..],0,"CHROME="); n = atp_put(env[..],n,cast(string)self[0..<(ifx sn > 0 then sn else 0)]); ok = proc_env(h,cast(string)env[0..<n]) && ok;
  n = atp_put(env[..],0,"ANIME_TAKE_CHROME_SENTINEL="); n = atp_put(env[..],n,atp_path(t,"unexpected-chrome-launch.txt",marker[..])); ok = proc_env(h,cast(string)env[0..<n]) && ok;
  started := ok && proc_start(h) == 0; line: [4096] u8; ln := ifx started then proc_line(h,atp_timeout(t,30000),line[..]) else -1;
  status := ifx started then proc_wait(h,1000) else -1; if h >= 0 { if status == PROC_RUNNING { killed := proc_kill(h,PROC_SIGKILL); } done := proc_release(h); }
  nodes: [64] JvNode; d: [1] Jv;
  refused := ln > 0 && jv_parse(d[..],cast(string)line[0..<ln],nodes[..]) && jv_kind(d[0],jv_get(d[0],0,"ok")) == JV_FALSE;
  written := atp_write(t,"combined-input-refusal.cli.json",ifx ln > 0 then cast(string)line[0..<ln] else "{}");
  atp_check(t,"oversized URL plus eval refuses usage2 before Chrome and output",written && started && status == 2 && refused && !fs_exists(atp_path(t,"refused.png",path[..])) && !fs_exists(atp_path(t,"unexpected-chrome-launch.txt",marker[..])));
}

tool_main :: () -> int #export_c {
  marker: [4096] u8; sentinel := proc_getenv("ANIME_TAKE_CHROME_SENTINEL",marker[..]);
  if sentinel != "" { wrote := fs_write(sentinel,"The CLI started Chrome before refusing its oversized input.\n"); return 99; }
  t: [1] Atp; cli: [4096] u8; base: [512] u8; dir: [4096] u8; t[0].end_ms = proc_now_ms()+780000;
  t[0].cli = proc_getenv("ANIME_TAKE_CLI",cli[..]); t[0].base = proc_getenv("ANIME_TAKE_BASE",base[..]); t[0].dir = proc_getenv("ANIME_TAKE_RECORD",dir[..]);
  if t[0].cli == "" || t[0].dir == "" || t[0].base.count < 18 || t[0].base[0..<17] != "http://127.0.0.1:" { tool_errln("set ANIME_TAKE_CLI, loopback ANIME_TAKE_BASE and ANIME_TAKE_RECORD"); return 2; }
  if !fs_mkdir(t[0].dir) { return 1; }
  mode_buf: [32] u8; mode := proc_getenv("ANIME_TAKE_ONLY",mode_buf[..]);
  if mode != "" && mode != "shots" && mode != "clips" && mode != "transport" && mode != "refusal" { return 2; }
  wrote := atp_write(t[..],"take.json",ATP_TAKE) && atp_write(t[..],"take-without-make.json",ATP_WITHOUT_MAKE);
  if !wrote { return 1; }
