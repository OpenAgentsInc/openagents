#!/usr/bin/env python3
"""Compile and run preserved ordinary Coax gameplay integration programs."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import platform
import re
import shutil
import signal
import subprocess
import tempfile
import time

ROOT = Path(__file__).resolve().parents[2]
MANIFEST = ROOT / 'aethyr/tests/native-gameplay-checks.json'
DRIVER = ROOT / 'aethyr/tests/native-gameplay-driver.c'


def digest(data):
    return hashlib.sha256(data).hexdigest()


def snapshot(entries):
    seen = {}

    def visit(path):
        path = path.resolve()
        if path in seen:
            return
        data = path.read_bytes()
        seen[path] = digest(data)
        if path.suffix == '.coax':
            for name in re.findall(rb'^\s*#(?:load|import_proto)\s+"([^"\n]+)"', data, re.M):
                visit(path.parent / name.decode())

    for path in entries:
        visit(path)
    return {str(path.relative_to(ROOT)): value for path, value in sorted(seen.items())}


def original_body(source, row):
    marker = row['procedure'] + ' :: () #export_c'
    if source.count(marker) != 1:
        raise RuntimeError('native procedure identity is not unique')
    first = source.index('{', source.index(marker) + len(marker))
    if row['kind'] == 'forall':
        bindings = row['original_bindings'].strip()
        if bindings != 'k: 0..<10' or row['cases'] != 10:
            raise RuntimeError('unsupported finite scenario domain')
        first = source.index('{', source.index('for ' + bindings, first))
    elif row['cases'] != 1:
        raise RuntimeError('ordinary scenario must have one original body')
    at = first
    depth = 0
    while at < len(source):
        if source.startswith('//', at):
            at = source.index('\n', at)
            continue
        if source.startswith('/*', at):
            at = source.index('*/', at + 2) + 2
            continue
        token = source[at]
        if token in ('"', "'"):
            quote = token
            at += 1
            while at < len(source):
                if source[at] == '\\':
                    at += 2
                elif source[at] == quote:
                    at += 1
                    break
                else:
                    at += 1
            continue
        if token == '{':
            depth += 1
        elif token == '}':
            depth -= 1
            if depth == 0:
                end = at + 1
                if source[end:end + 1] == '\n':
                    end += 1
                if digest(source[first:end].encode()) != row['body_sha256']:
                    raise RuntimeError('original body changed: ' + row['original_name'])
                return
        at += 1
    raise RuntimeError('unbalanced original body')


def qualify(args, work):
    manifest = json.loads(MANIFEST.read_text())
    records = [row for row in manifest['records'] if row['realm'] == args.realm]
    entries = {row['entry'] for row in records}
    if len(entries) != 1:
        raise RuntimeError('expected one independent realm entry')
    entry = entries.pop()
    for row in records:
        original_body((ROOT / entry).read_text(), row)
    coax_sources = snapshot([ROOT / entry])
    sources = snapshot([ROOT / entry, MANIFEST, DRIVER, Path(__file__)])
    compiler = Path(args.coax).resolve()
    compiler_hash = digest(compiler.read_bytes())
    cc = os.environ.get('CC', 'cc')
    cc_path = shutil.which(cc)
    native_version = subprocess.check_output([cc, '--version'], text=True, timeout=10)
    began = time.monotonic()
    deadline = began + 830
    commands = []
    outcomes = []
    emission = None
    receipt = {'schema': 'aethyr.native-gameplay-programs-execution.v1', 'mode': args.mode,
               'realm': args.realm, 'entry': entry, 'sources': sources,
               'compiler': str(compiler), 'compiler_sha256': compiler_hash,
               'native_compiler': cc_path, 'native_compiler_version': native_version,
               'platform': platform.platform(), 'native_stack_bytes': 128 * 1024 * 1024,
               'scope': 'Ordinary Coax integration procedures with original bodies/#assert statements. Normal authority checks remain enabled. No native-check ABI, evaluator-parity, Wasm-execution or raster claim.'}

    def save(status):
        after = snapshot([ROOT / entry, MANIFEST, DRIVER, Path(__file__)])
        changes = [name for name in sorted(set(sources) | set(after)) if sources.get(name) != after.get(name)]
        receipt.update(status=status, source_changes=changes, commands=commands, outcomes=outcomes,
                       emission=emission, elapsed_seconds=time.monotonic() - began)
        if args.receipt:
            Path(args.receipt).write_text(json.dumps(receipt, indent=2) + '\n')
        return changes

    def run(command, label, *, code_file=None, expected_assertion=False, limit=180):
        log = work / (label + '.log')
        timeout = min(limit, deadline - time.monotonic())
        if timeout <= 0:
            raise RuntimeError('integration command reached its 830-second limit')
        started = time.monotonic()
        with log.open('wb') as output:
            code_output = code_file.open('wb') if code_file else output
            try:
                process = subprocess.Popen(command, cwd=ROOT, stdout=code_output, stderr=output,
                                           start_new_session=True, env=dict(os.environ,
                                           ASAN_OPTIONS='detect_leaks=0:halt_on_error=1',
                                           UBSAN_OPTIONS='halt_on_error=1:print_stacktrace=1'))
                try:
                    result = process.wait(timeout=timeout)
                except subprocess.TimeoutExpired:
                    os.killpg(process.pid, signal.SIGKILL)
                    process.wait()
                    result = 124
            finally:
                if code_file:
                    code_output.close()
        output = log.read_text(errors='replace')
        commands.append({'command': list(map(str, command)), 'exit_code': result,
                         'expected_assertion_trap': expected_assertion,
                         'elapsed_seconds': time.monotonic() - started, 'timeout_seconds': timeout,
                         'log': str(log), 'log_sha256': digest(log.read_bytes())})
        passed = (result != 0 and result != 124 and 'assertion failed' in output) if expected_assertion else result == 0
        if not passed:
            save('failed')
            raise RuntimeError(f'{label} exited {result}\n{output[-6000:]}')
        return output

    mode = args.mode
    coax_flags = ['--release'] if mode == 'release' else []
    native_flags = {'debug': ['-O0'], 'release': ['-O2'],
                    'sanitized': ['-O1', '-fsanitize=address,undefined', '-fno-sanitize-recover=all',
                                  '-fno-omit-frame-pointer']}[mode]
    if args.reuse_emission:
        emission = json.loads(Path(args.reuse_emission).read_text())
        code = Path(emission['c_file'])
        if (emission['entry'] != entry or emission['compiler_sha256'] != compiler_hash or
                emission['coax_flags'] != coax_flags or emission['sources'] != coax_sources or
                digest(code.read_bytes()) != emission['c_sha256']):
            raise RuntimeError('C emission reuse does not match this source/compiler/profile')
        print('reusing qualified normal C emission: ' + str(code), flush=True)
    else:
        code = work / (args.realm + '-' + mode + '.c')
        print('normal Coax C emission: ' + entry + ' ' + mode, flush=True)
        run([str(compiler), 'c', entry, '--library', *coax_flags], 'emit', code_file=code, limit=810)
        emission = {'kind': 'normal Coax C library emission; authority enabled', 'entry': entry,
                    'compiler_sha256': compiler_hash, 'coax_flags': coax_flags, 'sources': coax_sources,
                    'c_file': str(code), 'c_sha256': digest(code.read_bytes()), 'command': commands[-1]}
        (work / 'emission.json').write_text(json.dumps(emission, indent=2) + '\n')
    registry = work / 'native-gameplay-registry.h'
    declarations = []
    values = []
    for row in records:
        name = row['procedure']
        if not re.fullmatch(r'aethyr_native_(everglade|deadgears)_[0-9]+', name):
            raise RuntimeError('invalid scenario symbol')
        declarations.append('extern void ' + name + '(void);')
        values.append('  {' + json.dumps(name) + ', ' + name + ', ' + str(row['cases']) + '},')
    registry.write_text('\n'.join(declarations) + '\nstatic const NativeScenario scenarios[] = {\n' + '\n'.join(values) + '\n};\n')
    binary = work / (args.realm + '-' + mode)
    run([cc, '-std=c11', *native_flags, '-ffp-contract=off', '-fno-fast-math', '-pthread',
         '-I' + str(work), str(code), str(DRIVER), '-lm', '-o', str(binary)], 'compile')
    run([str(binary), '--assert-probe'], 'assert-probe', expected_assertion=True)
    for index, row in enumerate(records):
        output = run([str(binary), row['procedure']], 'scenario-' + str(index))
        print(mode + ': ' + output, end='', flush=True)
        outcomes.append({'procedure': row['procedure'], 'original_name': row['original_name'],
                         'original_body_calls': row['cases'], 'body_sha256': row['body_sha256']})
    receipt['binary_sha256'] = digest(binary.read_bytes())
    if save('passed'):
        raise RuntimeError('sources changed during native integration qualification')
    print(f'{len(outcomes)} integration programs passed; {sum(row["original_body_calls"] for row in outcomes)} original body calls')


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--coax', default=str(ROOT / 'coax/bin/coax'))
    parser.add_argument('--realm', choices=['everglade', 'deadgears'], required=True)
    parser.add_argument('--mode', choices=['debug', 'release', 'sanitized'], default='debug')
    parser.add_argument('--reuse-emission', help='Reuse an identical qualified emission.json, e.g. debug C for sanitizers')
    parser.add_argument('--receipt')
    parser.add_argument('--work-dir')
    args = parser.parse_args()
    if args.work_dir:
        work = Path(args.work_dir).resolve()
        work.mkdir(parents=True, exist_ok=True)
        qualify(args, work)
    else:
        with tempfile.TemporaryDirectory(prefix='aethyr-native-gameplay-') as temporary:
            qualify(args, Path(temporary))


if __name__ == '__main__':
    main()
