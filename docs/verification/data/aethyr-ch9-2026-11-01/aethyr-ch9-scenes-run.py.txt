def snapshot():
    seen={}
    def visit(p):
        p=p.resolve()
        if not p.is_file() or p in seen: return
        data=p.read_bytes()
        seen[p]=hashlib.sha256(data).hexdigest()
        if p.suffix=='.coax':
            for dep in re.findall(rb'^\s*#(?:load|import_proto)\s+"([^"\n]+)"',data,re.M): visit(p.parent/dep.decode())
        elif p.suffix in ['.c','.h']:
            for dep in re.findall(rb'^\s*#include\s+"([^"\n]+)"',data,re.M): visit(p.parent/dep.decode())
    for r in roots: visit(root/r)
    return {str(k.relative_to(root)):v for k,v in sorted(seen.items())}
