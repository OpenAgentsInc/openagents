psf_applied :: (actuator:PidOutput,external:PidOutput,passive:PidOutput)->PidOutput {
 out:PidOutput;for i:0..<7 {driven:=actuator.hinge[i]+external.hinge[i];out.hinge[i]=driven+passive.hinge[i];}for i:0..<2 {driven:=actuator.slide[i]+external.slide[i];out.slide[i]=driven+passive.slide[i];}return out;
}

 if angles.count!=7 || slides.count!=2 || hinge_rates.count!=7 || slide_rates.count!=2 || arm_control.count!=7 || finger_target.count!=2 || external_hinge.count!=7 || external_slide.count!=2 || output.count!=1 {return PmsStatus.shape;}
